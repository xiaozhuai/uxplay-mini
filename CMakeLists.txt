cmake_minimum_required(VERSION 3.20)

# Never let a preset or an IDE install dependencies implicitly.
set(VCPKG_MANIFEST_INSTALL OFF CACHE BOOL "Dependency installation requires explicit approval")
project(uxplay_mini VERSION 0.1.1 LANGUAGES C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(WIN32)
    # Keep Windows headers from defining min/max macros that break C++ functions.
    add_compile_definitions(NOMINMAX)
endif()

include(CTest)
include(GNUInstallDirs)
include(CheckIncludeFile)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(llhttp CONFIG REQUIRED)
find_package(unofficial-libplist CONFIG REQUIRED)

option(UXPLAY_MINI_INTERNAL_MDNS "Use bundled mDNS instead of system Bonjour on macOS" OFF)
option(UXPLAY_MINI_WITH_AUDIO "Decode AAC/ALAC to L16 RTP using libavcodec and libavutil" ON)
option(UXPLAY_MINI_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)

set(core vendor/uxplay/lib)
add_library(airplay_core STATIC
    ${core}/byteutils.c ${core}/compat.c ${core}/crypto.c ${core}/dnssd.c
    ${core}/fairplay_playfair.c ${core}/httpd.c ${core}/http_request.c
    ${core}/http_response.c ${core}/logger.c ${core}/mirror_buffer.c
    ${core}/netutils.c ${core}/pairing.c ${core}/raop.c ${core}/raop_buffer.c
    ${core}/raop_ntp.c ${core}/raop_rtp.c ${core}/raop_rtp_mirror.c
    ${core}/srp.c ${core}/utils.c
    ${core}/playfair/hand_garble.c ${core}/playfair/modified_md5.c
    ${core}/playfair/omg_hax.c ${core}/playfair/playfair.c ${core}/playfair/sap_hash.c)
target_include_directories(airplay_core PUBLIC ${core} PRIVATE ${core}/playfair)
target_compile_definitions(airplay_core PRIVATE PLIST_210 PLIST_230 NOHOLD OPENSSL_API_COMPAT=0x10101000L)
target_link_libraries(airplay_core PUBLIC Threads::Threads PRIVATE OpenSSL::Crypto unofficial::libplist::libplist llhttp::llhttp_static)
check_include_file(sys/endian.h HAVE_SYS_ENDIAN_H)
if(HAVE_SYS_ENDIAN_H)
    target_compile_definitions(airplay_core PRIVATE SYS_ENDIAN_H)
endif()
if(APPLE AND NOT UXPLAY_MINI_INTERNAL_MDNS)
    target_sources(airplay_core PRIVATE ${core}/dns_sd/dns_sd.c)
    target_compile_definitions(airplay_core PRIVATE EXTERNAL_DNS_SD)
    # DNSService* is part of libSystem on macOS, linked by the compiler driver.
else()
    target_sources(airplay_core PRIVATE ${core}/mdnsd/mdnsd.c ${core}/mdnsd/dnssd_mdnsd.c)
endif()
if(WIN32)
    target_compile_definitions(airplay_core PUBLIC WIN32 _WIN32_WINNT=0x0601 _CRT_SECURE_NO_WARNINGS)
    target_link_libraries(airplay_core PUBLIC ws2_32 iphlpapi)
elseif(NOT APPLE)
    target_link_libraries(airplay_core PRIVATE m)
endif()

add_library(rtp STATIC src/rtp.cpp)
target_include_directories(rtp PUBLIC src)
if(WIN32)
    target_link_libraries(rtp PUBLIC ws2_32)
endif()

add_library(audio STATIC src/audio.cpp)
target_include_directories(audio PUBLIC src)
if(UXPLAY_MINI_WITH_AUDIO)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(AVCODEC REQUIRED IMPORTED_TARGET libavcodec>=59)
    pkg_check_modules(AVUTIL REQUIRED IMPORTED_TARGET libavutil>=57)
    target_compile_definitions(audio PRIVATE MINI_AUDIO_FFMPEG)
    target_link_libraries(audio PRIVATE PkgConfig::AVCODEC PkgConfig::AVUTIL)
endif()

add_executable(uxplay-mini src/main.cpp)
target_link_libraries(uxplay-mini PRIVATE airplay_core rtp audio)
if(UXPLAY_MINI_WITH_AUDIO)
    target_compile_definitions(uxplay-mini PRIVATE MINI_WITH_AUDIO)
endif()
if(MSVC)
    target_compile_options(uxplay-mini PRIVATE /utf-8 /W4)
else()
    foreach(target rtp audio uxplay-mini)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endforeach()
endif()
if(UXPLAY_MINI_SANITIZERS AND NOT MSVC)
    foreach(target airplay_core rtp audio uxplay-mini)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    endforeach()
    target_link_options(uxplay-mini PRIVATE -fsanitize=address,undefined)
endif()

if(BUILD_TESTING)
    add_executable(rtp_tests tests/rtp_tests.cpp)
    target_link_libraries(rtp_tests PRIVATE rtp)
    add_test(NAME rtp COMMAND rtp_tests)
    add_executable(audio_tests tests/audio_tests.cpp)
    target_link_libraries(audio_tests PRIVATE audio)
    if(UXPLAY_MINI_WITH_AUDIO)
        target_compile_definitions(audio_tests PRIVATE MINI_EXPECT_AUDIO)
    endif()
    add_test(NAME audio COMMAND audio_tests)
    add_test(NAME cli_help COMMAND uxplay-mini --help)
    set_tests_properties(cli_help PROPERTIES PASS_REGULAR_EXPRESSION "--h265")
    add_test(NAME cli_invalid_port COMMAND uxplay-mini --video-port 0)
    set_tests_properties(cli_invalid_port PROPERTIES WILL_FAIL TRUE)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        add_test(NAME cli_options COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_tests.py" "$<TARGET_FILE:uxplay-mini>" "$<BOOL:${UXPLAY_MINI_WITH_AUDIO}>")
        set_tests_properties(cli_options PROPERTIES TIMEOUT 30)
        add_executable(udp_sender tests/udp_sender.cpp)
        target_link_libraries(udp_sender PRIVATE rtp)
        add_test(NAME udp_smoke COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/udp_smoke.py" "$<TARGET_FILE:udp_sender>")
        set_tests_properties(udp_smoke PROPERTIES LABELS network TIMEOUT 20)
        add_test(NAME protocol_smoke COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/protocol_smoke.py" "$<TARGET_FILE:uxplay-mini>" "$<BOOL:${UXPLAY_MINI_WITH_AUDIO}>")
        add_test(NAME protocol_h265 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/protocol_smoke.py" "$<TARGET_FILE:uxplay-mini>" "$<BOOL:${UXPLAY_MINI_WITH_AUDIO}>" --h265)
        add_test(NAME protocol_no_audio COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/protocol_smoke.py" "$<TARGET_FILE:uxplay-mini>" "$<BOOL:${UXPLAY_MINI_WITH_AUDIO}>" --no-audio)
        add_test(NAME protocol_h265_no_audio COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/protocol_smoke.py" "$<TARGET_FILE:uxplay-mini>" "$<BOOL:${UXPLAY_MINI_WITH_AUDIO}>" --h265 --no-audio)
        set_tests_properties(protocol_smoke protocol_h265 protocol_no_audio protocol_h265_no_audio PROPERTIES LABELS network TIMEOUT 30)
        if(APPLE)
            find_program(DNS_SD_CLI dns-sd)
            if(DNS_SD_CLI)
                add_test(NAME discovery_smoke COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/discovery_smoke.py" "$<TARGET_FILE:uxplay-mini>" "${DNS_SD_CLI}")
                add_test(NAME discovery_h265 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/discovery_smoke.py" "$<TARGET_FILE:uxplay-mini>" "${DNS_SD_CLI}" --h265)
                add_test(NAME discovery_no_audio COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/discovery_smoke.py" "$<TARGET_FILE:uxplay-mini>" "${DNS_SD_CLI}" --no-audio)
                add_test(NAME discovery_h265_no_audio COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/discovery_smoke.py" "$<TARGET_FILE:uxplay-mini>" "${DNS_SD_CLI}" --h265 --no-audio)
                set_tests_properties(discovery_smoke discovery_h265 discovery_no_audio discovery_h265_no_audio PROPERTIES LABELS network TIMEOUT 25)
            endif()
        endif()
    endif()
    if(UXPLAY_MINI_SANITIZERS AND NOT MSVC)
        set(test_targets rtp_tests audio_tests)
        if(TARGET udp_sender)
            list(APPEND test_targets udp_sender)
        endif()
        foreach(target IN LISTS test_targets)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endforeach()
        get_property(all_tests DIRECTORY PROPERTY TESTS)
        set_tests_properties(${all_tests} PROPERTIES ENVIRONMENT "UBSAN_OPTIONS=halt_on_error=1:print_stacktrace=1")
    endif()
endif()

install(TARGETS uxplay-mini RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES LICENSE README.md README_CN.md UPSTREAM.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(FILES ${core}/playfair/LICENSE.md DESTINATION ${CMAKE_INSTALL_DOCDIR}/playfair)
